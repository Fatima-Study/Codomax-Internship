# Day 4 - Python Basics

# Variables
name = "Fatima"
age = 20
department = "Telecommunication Engineering"

print("Name:", name)
print("Age:", age)
print("Department:", department)


# List
subjects = ["Python", "AI", "Machine Learning", "Cybersecurity"]

print("\nSubjects:")
print(subjects)


# Loop
print("\nSubjects using loop:")

for subject in subjects:
    print("-", subject)


# Function
def greet(name):
    print("\nHello", name)
    print("Welcome to Python Programming!")


greet(name)
